// use task6

use task6

// 1) Explicit collection "books" with validation (non-empty title)
db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: { bsonType: "string", minLength: 1, description: "title must be a non-empty string" }
      }
    }
  }
})

// 2) Implicit collection "authors"
db.authors.insertOne({ name: "Author1", nationality: "British" })

// 3) Capped collection "logs" (1MB)
db.createCollection("logs", { capped: true, size: 1048576 })

// 4) Index on books.title
db.books.createIndex({ title: 1 })

// 5) Insert one book
db.books.insertOne({ title: "Book1", author: "Ali", year: 1937, genres: ["Fantasy", "Adventure"] })

// 6) Insert many books (>= 3)
db.books.insertMany([
  { title: "Future", author: "George Orwell", year: 2020, genres: ["Science Fiction"] },
  { title: "To Kill a Mockingbird", author: "Harper Lee", year: 1960, genres: ["Classic", "Fiction"] },
  { title: "Brave New World", author: "Aldous Huxley", year: 2006, genres: ["Dystopian", "Science Fiction"] }
])

// 7) Insert a log (use a real book _id so the $lookup in Q19 works)
const futureBook = db.books.findOne({ title: "Future" })
db.logs.insertOne({ book_id: futureBook._id, action: "borrowed" })

// 8) Update "Future" -> year 2022
db.books.updateOne({ title: "Future" }, { $set: { year: 2022 } })

// 9) Find book with title "Brave New World"
db.books.findOne({ title: "Brave New World" })

// 10) Books published between 1990 and 2010
db.books.find({ year: { $gte: 1990, $lte: 2010 } })

// 11) Books where genres include "Science Fiction"
db.books.find({ genres: "Science Fiction" })

// 12) Skip first 2, limit next 3, sort by year desc
db.books.find().sort({ year: -1 }).skip(2).limit(3)

// 13) Books where year is stored as integer
db.books.find({ year: { $type: "int" } })
// (mongosh stores plain numbers as double; use NumberInt(1937) when inserting
//  or use: { year: { $type: "number" } } to match any numeric type)

// 14) Books whose genres include neither "Horror" nor "Science Fiction"
db.books.find({ genres: { $nin: ["Horror", "Science Fiction"] } })

// 15) Delete all books published before 2000
db.books.deleteMany({ year: { $lt: 2000 } })

// 16) Aggregation: year > 2000, sorted by year desc
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $sort: { year: -1 } }
])

// 17) Aggregation: year > 2000, show only title, author, year
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $project: { _id: 0, title: 1, author: 1, year: 1 } }
])

// 18) Aggregation: unwind genres
db.books.aggregate([
  { $unwind: "$genres" },
  { $project: { _id: 0, title: 1, genres: 1 } }
])

// 19) Aggregation: join logs with books
db.logs.aggregate([
  { $lookup: { from: "books", localField: "book_id", foreignField: "_id", as: "book_details" } },
  { $project: { _id: 0, action: 1, "book_details.title": 1, "book_details.author": 1, "book_details.year": 1 } }
])
